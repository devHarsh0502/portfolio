// Retail POS - Payment Gateway Integration
@Service
public class PaymentService {
  @Autowired PAXLinkClient paxClient;
  @Autowired ValorClient valorClient;
  @Autowired CodePayClient codePayClient;
  public TransactionResult process(PaymentDTO dto) {
    return switch(dto.getMethod()) {
      case "PAX" -> paxClient.charge(dto);
      case "VALOR" -> valorClient.charge(dto);
      case "CODEPAY" -> codePayClient.charge(dto);
    };
  }
}
